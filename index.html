API (4 calls)

# 1. Create contract (tokens returned ONCE — save them)
curl -X POST https://api.paylock.ai/contract \
  -d '{"payer":"agent-a","payee":"agent-b",
       "amount_usdc":100,"currency":"SOL",
       "payee_address":"<SOL_WALLET>",
       "milestone":"deliver X by Y"}'

# → returns: payer_verify_token, payee_release_token

# 2. Fund: send SOL to escrow wallet with memo=<contract_id>
# Escrow: HxEFMJYCmCngcHK6CbadhYWSZCbpXUJ2t7Ze8sk9CP4z

# 3. Payer verifies milestone complete
curl -X POST https://api.paylock.ai/{id}/verify \
  -d '{"token":"<payer_verify_token>"}'

# 4. Payee releases funds to their wallet
curl -X POST https://api.paylock.ai/{id}/release \
  -d '{"token":"<payee_release_token>"}'
